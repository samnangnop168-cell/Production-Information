        Google Apps Script — Shift & Downtime Logger

        Paste into Extensions → Apps Script on a Google Sheet to log entries. 

        const SHEET_NAME = 'ShiftDowntimeLog';

function doPost(e) {
  try {
    const data = JSON.parse(e.postData.contents);
    const sheet = getOrCreateLogSheet_();

    const startMins = toMinutes_(data.timeStart);
    const endMins = toMinutes_(data.timeEnd);
    const computedDuration =
      typeof data.durationMinutes === 'number'
        ? data.durationMinutes
        : ((endMins - startMins + 1440) % 1440);

    sheet.appendRow([
      new Date(),
      data.date || '',
      data.line || 'Line A',
      data.modelName || 'Unspecified',
      data.timeStart || '',
      data.timeEnd || '',
      (computedDuration / 60).toFixed(2),
      data.reasonStop || 'None'
    ]);

    return ContentService
      .createTextOutput(JSON.stringify({ status: 'ok', message: 'Record logged' }))
      .setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService
      .createTextOutput(JSON.stringify({ status: 'error', message: err.message }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}

function getOrCreateLogSheet_() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  let sheet = ss.getSheetByName(SHEET_NAME);
  if (!sheet) {
    sheet = ss.insertSheet(SHEET_NAME);
    sheet.appendRow([
      'Logged At', 'Date', 'Line', 'Model Name', 'Time Start', 'Time End', 'Duration (Hrs)', 'Reason Stop'
    ]);
    sheet.setFrozenRows(1);
  }
  return sheet;
}

function toMinutes_(hhmm) {
  const [h, m] = String(hhmm || '00:00').split(':').map(Number);
  return ((h || 0) * 60) + (m || 0);
}
